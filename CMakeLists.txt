cmake_minimum_required(VERSION 3.24)
project(AutoHotInterceptor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(InterceptorBridge SHARED Native/Bridge.cpp Native/SDK/Interceptor.cpp)
target_compile_definitions(InterceptorBridge PRIVATE INTERCEPTION_EXPORT UNICODE _UNICODE NOMINMAX)
target_link_libraries(InterceptorBridge PRIVATE setupapi cfgmgr32 hid advapi32 bcrypt)
target_compile_options(InterceptorBridge PRIVATE /W4 /EHsc /permissive- /utf-8)
include(CTest)
if(BUILD_TESTING)
  add_library(InterceptorBridgeFixture SHARED Tests/FakeSdk.cpp Native/Bridge.cpp)
  target_compile_definitions(InterceptorBridgeFixture PRIVATE INTERCEPTION_EXPORT UNICODE _UNICODE NOMINMAX)
  target_compile_options(InterceptorBridgeFixture PRIVATE /W4 /EHsc /utf-8)
  set_target_properties(InterceptorBridgeFixture PROPERTIES OUTPUT_NAME InterceptorBridge RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Fixture")
  add_executable(BridgeTests Tests/BridgeTests.cpp Tests/FakeSdk.cpp Native/Bridge.cpp)
  target_compile_definitions(BridgeTests PRIVATE INTERCEPTION_STATIC UNICODE _UNICODE NOMINMAX)
  target_compile_options(BridgeTests PRIVATE /W4 /EHsc /utf-8)
  add_test(NAME BridgeTests COMMAND BridgeTests)
endif()
